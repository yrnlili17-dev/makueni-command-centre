async function ensureExecutiveDecisionTables() {
  await db.execute(sql`
    CREATE TABLE IF NOT EXISTS executive_decisions (
      id bigserial PRIMARY KEY,
      title text NOT NULL,
      decision_type text NOT NULL DEFAULT 'operational',
      priority text NOT NULL DEFAULT 'high',
      status text NOT NULL DEFAULT 'pending',
      owner text,
      due_date date,
      rationale text,
      source_module text,
      metadata jsonb NOT NULL DEFAULT '{}'::jsonb,
      created_at timestamptz NOT NULL DEFAULT now(),
      updated_at timestamptz NOT NULL DEFAULT now()
    );

    CREATE TABLE IF NOT EXISTS executive_audit_log (
      id bigserial PRIMARY KEY,
      action text NOT NULL,
      entity_type text NOT NULL,
      entity_id text,
      actor text,
      details jsonb NOT NULL DEFAULT '{}'::jsonb,
      created_at timestamptz NOT NULL DEFAULT now()
    );
  `);
}

router.get("/executive-decisions", async (_req, res) => {
  try {
    await ensureExecutiveDecisionTables();

    const result = await db.execute(sql`
      SELECT
        id,
        title,
        decision_type AS "decisionType",
        priority,
        status,
        owner,
        due_date AS "dueDate",
        rationale,
        source_module AS "sourceModule",
        metadata,
        created_at AS "createdAt",
        updated_at AS "updatedAt"
      FROM executive_decisions
      ORDER BY
        CASE priority
          WHEN 'critical' THEN 1
          WHEN 'high' THEN 2
          WHEN 'medium' THEN 3
          ELSE 4
        END,
        CASE status
          WHEN 'pending' THEN 1
          WHEN 'approved' THEN 2
          WHEN 'in_progress' THEN 3
          WHEN 'deferred' THEN 4
          ELSE 5
        END,
        updated_at DESC
      LIMIT 100
    `);

    res.json((result as any).rows ?? []);
  } catch (err) {
    console.error("executive decisions list failed", err);
    res.status(500).json({
      error: "Failed to load executive decisions",
      detail: err instanceof Error ? err.message : "Unknown decisions error",
    });
  }
});

router.post("/executive-decisions", async (req, res) => {
  try {
    await ensureExecutiveDecisionTables();

    const body = req.body as any;
    const title = String(body?.title ?? "").trim();

    if (!title) {
      res.status(400).json({ error: "title required" });
      return;
    }

    const result = await db.execute(sql`
      INSERT INTO executive_decisions (
        title,
        decision_type,
        priority,
        status,
        owner,
        due_date,
        rationale,
        source_module,
        metadata
      )
      VALUES (
        ${title},
        ${body?.decisionType ?? "operational"},
        ${body?.priority ?? "high"},
        ${body?.status ?? "pending"},
        ${body?.owner ?? null},
        ${body?.dueDate ?? null},
        ${body?.rationale ?? null},
        ${body?.sourceModule ?? "executive-command"},
        ${JSON.stringify(body?.metadata ?? {})}::jsonb
      )
      RETURNING
        id,
        title,
        decision_type AS "decisionType",
        priority,
        status,
        owner,
        due_date AS "dueDate",
        rationale,
        source_module AS "sourceModule",
        metadata,
        created_at AS "createdAt",
        updated_at AS "updatedAt"
    `);

    const decision = (result as any).rows?.[0];

    await db.execute(sql`
      INSERT INTO executive_audit_log (
        action,
        entity_type,
        entity_id,
        actor,
        details
      )
      VALUES (
        'create',
        'executive-decision',
        ${String(decision?.id ?? "")},
        ${body?.actor ?? "system"},
        ${JSON.stringify({
          title,
          priority: body?.priority ?? "high",
          sourceModule: body?.sourceModule ?? "executive-command",
        })}::jsonb
      )
    `);

    res.status(201).json(decision);
  } catch (err) {
    console.error("executive decision create failed", err);
    res.status(500).json({
      error: "Failed to create executive decision",
      detail: err instanceof Error ? err.message : "Unknown create error",
    });
  }
});

router.patch("/executive-decisions/:id", async (req, res) => {
  try {
    await ensureExecutiveDecisionTables();

    const id = Number(req.params.id);
    if (!Number.isInteger(id)) {
      res.status(400).json({ error: "Invalid decision id" });
      return;
    }

    const body = req.body as any;

    const result = await db.execute(sql`
      UPDATE executive_decisions
      SET
        title = coalesce(${body?.title ?? null}, title),
        decision_type = coalesce(${body?.decisionType ?? null}, decision_type),
        priority = coalesce(${body?.priority ?? null}, priority),
        status = coalesce(${body?.status ?? null}, status),
        owner = coalesce(${body?.owner ?? null}, owner),
        due_date = coalesce(${body?.dueDate ?? null}, due_date),
        rationale = coalesce(${body?.rationale ?? null}, rationale),
        updated_at = now()
      WHERE id = ${id}
      RETURNING
        id,
        title,
        decision_type AS "decisionType",
        priority,
        status,
        owner,
        due_date AS "dueDate",
        rationale,
        source_module AS "sourceModule",
        metadata,
        created_at AS "createdAt",
        updated_at AS "updatedAt"
    `);

    if ((result as any).rows?.length === 0) {
      res.status(404).json({ error: "Decision not found" });
      return;
    }

    await db.execute(sql`
      INSERT INTO executive_audit_log (
        action,
        entity_type,
        entity_id,
        actor,
        details
      )
      VALUES (
        'update',
        'executive-decision',
        ${String(id)},
        ${body?.actor ?? "system"},
        ${JSON.stringify(body ?? {})}::jsonb
      )
    `);

    res.json((result as any).rows[0]);
  } catch (err) {
    console.error("executive decision update failed", err);
    res.status(500).json({
      error: "Failed to update executive decision",
      detail: err instanceof Error ? err.message : "Unknown update error",
    });
  }
});

router.post("/executive-decisions/generate", async (_req, res) => {
  try {
    await ensureExecutiveDecisionTables();

    const overview = await db.execute(sql`
      SELECT
        count(*)::integer AS constituents,
        count(*) FILTER (
          WHERE phone IS NULL OR btrim(phone) = ''
        )::integer AS missing_phone,
        count(*) FILTER (
          WHERE support_level IS NULL OR btrim(support_level) = ''
        )::integer AS missing_support,
        count(DISTINCT ward) FILTER (
          WHERE ward IS NOT NULL AND btrim(ward) <> ''
        )::integer AS wards
      FROM campaign_constituents
    `);

    const row = (overview as any).rows?.[0] ?? {};
    const constituents = Number(row.constituents ?? 0);
    const missingPhone = Number(row.missing_phone ?? 0);
    const missingSupport = Number(row.missing_support ?? 0);
    const wards = Number(row.wards ?? 0);

    const recommendations = [
      {
        title: `Recover ${missingPhone.toLocaleString()} missing phone contacts`,
        decisionType: "data",
        priority: missingPhone > constituents * 0.4 ? "critical" : "high",
        rationale:
          "Direct voter contact and mobilisation remain constrained by missing phone data.",
        sourceModule: "campaign-database",
      },
      {
        title: `Classify support for ${missingSupport.toLocaleString()} constituent records`,
        decisionType: "intelligence",
        priority: missingSupport > constituents * 0.5 ? "critical" : "high",
        rationale:
          "Persuasion and mobilisation planning require reliable support classification.",
        sourceModule: "campaign-database",
      },
      {
        title: `Assign ward execution owners across ${wards} imported wards`,
        decisionType: "operations",
        priority: "high",
        rationale:
          "County execution requires clear ownership, deadlines and measurable ward outputs.",
        sourceModule: "war-room",
      },
      {
        title: "Review open threats and assign response owners",
        decisionType: "intelligence",
        priority: "critical",
        rationale:
          "Narrative and operational threats should not remain unassigned.",
        sourceModule: "intelligence",
      },
      {
        title: "Activate measurable volunteer and field deployment",
        decisionType: "field",
        priority: "high",
        rationale:
          "Campaign readiness depends on visible activity and tracked household engagement.",
        sourceModule: "operations-hub",
      },
    ];

    let created = 0;

    for (const item of recommendations) {
      const exists = await db.execute(sql`
        SELECT id
        FROM executive_decisions
        WHERE title = ${item.title}
          AND status NOT IN ('completed', 'archived')
        LIMIT 1
      `);

      if ((exists as any).rows?.length > 0) continue;

      await db.execute(sql`
        INSERT INTO executive_decisions (
          title,
          decision_type,
          priority,
          status,
          rationale,
          source_module
        )
        VALUES (
          ${item.title},
          ${item.decisionType},
          ${item.priority},
          'pending',
          ${item.rationale},
          ${item.sourceModule}
        )
      `);

      created += 1;
    }

    await db.execute(sql`
      INSERT INTO executive_audit_log (
        action,
        entity_type,
        actor,
        details
      )
      VALUES (
        'generate',
        'executive-decision',
        'system',
        ${JSON.stringify({
          created,
          generatedAt: new Date().toISOString(),
        })}::jsonb
      )
    `);

    res.status(201).json({
      ok: true,
      created,
    });
  } catch (err) {
    console.error("executive decision generation failed", err);
    res.status(500).json({
      error: "Failed to generate executive decisions",
      detail: err instanceof Error ? err.message : "Unknown generation error",
    });
  }
});
